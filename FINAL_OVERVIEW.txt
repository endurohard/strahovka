⏰ АЛГОРИТМ РАБОТЫ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. Excel → Парсинг данных
  2. Расчет дат:
     • Дата окончания = Дата начала + 1 год
     • Дата напоминания = Дата окончания - 7 дней
  3. Сохранение в PostgreSQL
  4. Ежедневная проверка (10:00):
     • Поиск клиентов с reminder_date = сегодня
     • Отправка через WhatsApp
     • Отметка в БД

🚀 БЫСТРЫЙ СТАРТ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. Запуск:
     docker-compose up -d

  2. Авторизация WhatsApp:
     docker-compose logs -f app
     (отсканируйте QR код)

  3. Импорт данных:
     docker-compose exec app node index.js --import

  4. Проверка:
     ./check-db.sh

  5. Тест отправки:
     docker-compose exec app node index.js --check-now
